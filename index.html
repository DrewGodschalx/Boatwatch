Shared storage

This app is preconfigured to use the same Supabase database on every device.

Run/update this in Supabase → SQL Editor
create extension if not exists pgcrypto;

create table if not exists public.boats (
  id uuid primary key default gen_random_uuid(),
  name text not null,
  name_key text not null unique,
  boat_type text not null default 'Unknown',
  ns_sail_number text,
  mg_sail_number text,
  hull_type text not null default 'Unknown',
  location text not null default 'Unknown',
  lat double precision,
  lon double precision,
  owners jsonb not null default '[]'::jsonb,
  track jsonb not null default '[]'::jsonb,
  updated_at timestamptz not null default now()
);

alter table public.boats add column if not exists ns_sail_number text;
alter table public.boats add column if not exists mg_sail_number text;

alter table public.boats enable row level security;

drop policy if exists "Anyone can view boats" on public.boats;
drop policy if exists "Anyone can add boats" on public.boats;
drop policy if exists "Anyone can update boats" on public.boats;

create policy "Anyone can view boats"
  on public.boats for select using (true);
create policy "Anyone can add boats"
  on public.boats for insert with check (true);
create policy "Anyone can update boats"
  on public.boats for update using (true) with check (true);
These open policies allow anyone with access to this page to read and submit boat records. The app uses a public publishable/anon key, which is visible in the page source; never put a service-role key here. Boat records are shared through Supabase. Protect the project with backups and monitoring.
